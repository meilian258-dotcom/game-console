cmake_minimum_required(VERSION 3.24)
project(piq_libretro_jni LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "This experimental bridge supports Windows x64 only")
endif()
find_package(JNI REQUIRED COMPONENTS JVM)
add_library(piq-libretro-jni SHARED piq_libretro_jni.cpp)
target_compile_features(piq-libretro-jni PRIVATE cxx_std_20)
target_include_directories(piq-libretro-jni PRIVATE ${JNI_INCLUDE_DIRS})
target_link_libraries(piq-libretro-jni PRIVATE opengl32 gdi32 user32 bcrypt)
if(MSVC)
  set_property(TARGET piq-libretro-jni PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  target_compile_options(piq-libretro-jni PRIVATE /W4 /utf-8 /EHsc)
  target_link_options(piq-libretro-jni PRIVATE /Brepro)
else()
  target_compile_options(piq-libretro-jni PRIVATE -Wall -Wextra)
  target_link_options(piq-libretro-jni PRIVATE -static -Wl,--no-insert-timestamp)
  set_target_properties(piq-libretro-jni PROPERTIES PREFIX "")
endif()
